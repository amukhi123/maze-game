cmake_minimum_required(VERSION 3.23)

set(CMAKE_CXX_STANDARD 23)

project("MazeGame")

if(CMAKE_GENERATOR STREQUAL "Ninja")
    set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
endif()

file(GLOB_RECURSE SRC_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
file(GLOB_RECURSE INTERNAL_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h")

set(EXTERNAL_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/lib")
set(INTERNAL_HEADER_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/include")

add_executable(${PROJECT_NAME} ${SRC_FILES} ${INTERNAL_HEADERS})

source_group(
    TREE "${CMAKE_CURRENT_SOURCE_DIR}"
    FILES ${SRC_FILES}
)

source_group(
    TREE "${CMAKE_CURRENT_SOURCE_DIR}"
    FILES ${INTERNAL_HEADERS}
)

target_include_directories(
    ${PROJECT_NAME} PUBLIC
    ${INTERNAL_HEADER_DIRECTORY}
    ${EXTERNAL_HEADERS}
)

target_link_libraries(${PROJECT_NAME} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/lib/raylib/raylibdll.lib")
